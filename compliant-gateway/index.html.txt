<script>
    // Initialize the gateway using your official Merchant Public API Key
    const stripe = Stripe('pk_live_your_merchant_public_key_placeholder');
    const elements = stripe.elements();

    // Create and style the secure card input element
    const cardElement = elements.create('card', {
        style: {
            base: { fontSize: '16px', color: '#32325d', '::placeholder': { color: '#aab7c4' } }
        }
    });

    // Mount the secure input field into our DOM container
    cardElement.mount('#card-element');

    const form = document.getElementById('payment-form');
    form.addEventListener('submit', async (event) => {
        event.preventDefault();
        document.getElementById('submit-button').disabled = true;

        // Tokenize the card details directly on the provider's secure server
        const { token, error } = await stripe.createToken(cardElement, {
            name: document.getElementById('cardholder-name').value
        });

        if (error) {
            document.getElementById('error-message').textContent = error.message;
            document.getElementById('submit-button').disabled = false;
        } else {
            // Send the safe token string to your server backend
            sendTokenToServer(token.id);
        }
    });

    async function sendTokenToServer(tokenId) {
        // Send an un-hackable reference token to your backend API for clearing
        const response = await fetch('/api/process-charge', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ token: tokenId })
        });
        
        if (response.ok) {
            alert('Payment completed successfully via secure tokenization!');
        } else {
            alert('Payment execution failed on the server side.');
        }
    }
</script>
